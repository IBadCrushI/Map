<script>
    // Инициализация Cesium
    const viewer = new Cesium.Viewer('cesiumContainer', {
        terrainProvider: Cesium.createWorldTerrain()
    });

    const fileInput = document.getElementById('upload');

    fileInput.addEventListener('change', (e) => {
        const file = e.target.files[0];
        const reader = new FileReader();

        reader.onload = (evt) => {
            const data = new Uint8Array(evt.target.result);
            const workbook = XLSX.read(data, {type: 'array'});
            const sheetName = workbook.SheetNames[0];
            const worksheet = workbook.Sheets[sheetName];
            const jsonData = XLSX.utils.sheet_to_json(worksheet);

            plotPipeline(jsonData);
        };

        reader.readAsArrayBuffer(file);
    });

    function plotPipeline(pipelineData) {
        viewer.entities.removeAll();

        // Находим min и max для нормализации
        const minGradient = Math.min(...pipelineData.map(p => p.Gradient));
        const maxGradient = Math.max(...pipelineData.map(p => p.Gradient));
        const minPotential = Math.min(...pipelineData.map(p => p.Potential));
        const maxPotential = Math.max(...pipelineData.map(p => p.Potential));

        function getColor(gradient) {
            const t = (gradient - minGradient) / (maxGradient - minGradient);
            return Cesium.Color.BLUE.lerp(Cesium.Color.RED, t, new Cesium.Color());
        }

        // Линия газопровода
        const positions = pipelineData.map(p => Cesium.Cartesian3.fromDegrees(p.Longitude, p.Latitude, 0));
        viewer.entities.add({
            name: "Gas Pipeline",
            polyline: { positions: positions, width: 5, material: Cesium.Color.YELLOW }
        });

        // Цилиндры и подписи
        pipelineData.forEach(p => {
            const height = 20 + 80 * (p.Potential - minPotential) / (maxPotential - minPotential);
            viewer.entities.add({
                position: Cesium.Cartesian3.fromDegrees(p.Longitude, p.Latitude, 0),
                cylinder: { length: height, topRadius: 5, bottomRadius: 5, material: getColor(p.Gradient) }
            });

            viewer.entities.add({
                position: Cesium.Cartesian3.fromDegrees(p.Longitude, p.Latitude, height + 5),
                label: {
                    text: p.Potential.toString(),
                    font: '16px sans-serif',
                    fillColor: Cesium.Color.WHITE,
                    outlineColor: Cesium.Color.BLACK,
                    outlineWidth: 2,
                    style: Cesium.LabelStyle.FILL_AND_OUTLINE,
                    verticalOrigin: Cesium.VerticalOrigin.BOTTOM
                }
            });
        });

        viewer.zoomTo(viewer.entities);
    }
</script>
